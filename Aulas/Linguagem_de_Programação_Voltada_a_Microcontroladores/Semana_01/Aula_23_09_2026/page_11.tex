\begin{frame}[fragile]{Declaração e inicialização}
\begin{lstlisting}
int ciclos = 0;
double pressao_bar = 2.4;
\end{lstlisting}
\begin{center}
\begin{tabular}{lll}
\toprule
Tipo & Nome & Valor inicial \\
\midrule
\texttt{int} & \texttt{ciclos} & \texttt{0} \\
\texttt{double} & \texttt{pressao\_bar} & \texttt{2.4} \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}
\item Declarar informa ao compilador o nome e o tipo.
\item Inicializar estabelece o primeiro valor.
\item O ponto e vírgula encerra a declaração.
\end{itemize}
\end{frame}
