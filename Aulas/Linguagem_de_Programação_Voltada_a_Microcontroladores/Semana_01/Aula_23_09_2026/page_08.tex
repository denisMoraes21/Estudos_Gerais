\begin{frame}[fragile]{Caracteres e valores lógicos}
\begin{lstlisting}
#include <stdbool.h>

char setor = 'A';
bool disponivel = true;
bool alarme = false;
\end{lstlisting}
\begin{itemize}
\item \texttt{'A'} usa aspas simples: é um caractere.
\item \texttt{"A"} usa aspas duplas: é uma string, tema da próxima semana.
\item \texttt{false} corresponde a 0; \texttt{true}, a 1.
\item Um estado lógico descreve uma condição; o acionamento de um pino físico será estudado depois.
\end{itemize}
\end{frame}
