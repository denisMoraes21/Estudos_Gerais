\begin{frame}{Objetivos e percurso da aula}
\begin{block}{Ao final, você deverá conseguir}
Escolher tipos para representar informações de automação, declarar variáveis e acompanhar seus valores em um programa C.
\end{block}
\begin{enumerate}
\item Reconhecer tipos primitivos e suas características.
\item Declarar, inicializar e atualizar variáveis.
\item Exibir os valores e interpretar um rastreamento.
\item Resolver uma atividade guiada e quatro exercícios.
\end{enumerate}
\end{frame}
