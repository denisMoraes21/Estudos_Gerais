\begin{frame}[fragile]{Atribuição: atualizar o valor armazenado}
\begin{lstlisting}
int ciclos = 0;
ciclos = 12;
ciclos = 18;
\end{lstlisting}
\begin{center}
\begin{tabular}{lc}
\toprule
Após executar & Valor de \texttt{ciclos} \\
\midrule
\texttt{int ciclos = 0;} & 0 \\
\texttt{ciclos = 12;} & 12 \\
\texttt{ciclos = 18;} & 18 \\
\bottomrule
\end{tabular}
\end{center}
\texttt{=} armazena um valor. Testar igualdade usa \texttt{==}, assunto da próxima aula.
\end{frame}
