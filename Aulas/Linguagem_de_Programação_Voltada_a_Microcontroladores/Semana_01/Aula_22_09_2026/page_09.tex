\begin{frame}[fragile,t]{Pseudocódigo sequencial: média de leituras}
\begin{lstlisting}[language={}]
INICIO
  LER leitura1
  LER leitura2
  media <- (leitura1 + leitura2) / 2
  ESCREVER media
FIM
\end{lstlisting}
\begin{itemize}
\item Neste pseudocódigo, a divisão admite parte decimal.
\item Para 20 e 25, a média é 22,5.
\item Os parênteses indicam que a soma ocorre antes da divisão.
\end{itemize}
\small A divisão em C dependerá dos tipos dos operandos; veremos isso nas próximas aulas.
\end{frame}
