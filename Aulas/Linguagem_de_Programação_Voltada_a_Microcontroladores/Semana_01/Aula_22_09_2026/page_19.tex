\begin{frame}[fragile,t]{Primeiro contato com C}
\begin{lstlisting}
#include <stdio.h>

int main(void) {
    printf("Inicio do ciclo de leitura\n");
    printf("Fim do ciclo de leitura\n");
    return 0;
}
\end{lstlisting}
\begin{itemize}
\item \texttt{main}: ponto de entrada do programa no computador.
\item \texttt{printf}: escreve texto; \texttt{\textbackslash n}: muda de linha.
\item \texttt{return 0}: informa que o programa terminou com sucesso.
\end{itemize}
\small É um exemplo sequencial. Variáveis, expressões e decisões serão construídas nas próximas aulas.
\end{frame}
