\begin{frame}[fragile]{Implementação corrigida e contrato}
\begin{lstlisting}
size_t contar_no_limite(const int valores[],
                       size_t quantidade, int limite)
{
    size_t total = 0;
    for (size_t i = 0; i < quantidade; ++i) {
        if (valores[i] >= limite) ++total;
    }
    return total;
}
\end{lstlisting}
\begin{itemize}
\item Para quantidade positiva, o vetor deve conter esse número de elementos legíveis.
\item Para quantidade zero, nenhum elemento é acessado e o resultado é zero; aceita-se \texttt{NULL} nesse caso.
\item A função não modifica o vetor nem realiza entrada/saída.
\end{itemize}
\end{frame}
