\begin{frame}{7.1 — Um procedimento de investigação}
\begin{enumerate}
\item Escreva a regra e o resultado esperado.
\item Registre os dados que reproduzem a falha.
\item Reduza o cenário sem eliminar o problema.
\item Observe valores e formule uma hipótese.
\item Altere uma causa por vez.
\item Execute novamente e acrescente um teste de regressão.
\end{enumerate}
\begin{block}{Registro mínimo}
Versão do código, comando, entrada, esperado, observado e alteração que resolveu a falha.
\end{block}
\end{frame}
