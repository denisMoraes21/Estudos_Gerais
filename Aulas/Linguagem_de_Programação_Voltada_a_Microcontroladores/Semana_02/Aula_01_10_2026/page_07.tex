\begin{frame}[t]{8.2 GPIO: General Purpose Input/Output}
\begin{columns}[T]
\begin{column}{0.48\textwidth}\begin{block}{Entrada digital}
Lê um nível lógico externo. Exemplos: botão, contato e saída digital de um sensor compatível.
\end{block}\end{column}
\begin{column}{0.48\textwidth}\begin{block}{Saída digital}
Comanda um nível lógico. Exemplo desta aula: LED integrado da placa.
\end{block}\end{column}
\end{columns}
\medskip
\begin{itemize}
\item \texttt{LOW} e \texttt{HIGH} representam níveis lógicos.
\item Esses níveis correspondem a faixas de tensão definidas pelo hardware.
\item A direção e a função do pino precisam ser configuradas.
\end{itemize}
\end{frame}
