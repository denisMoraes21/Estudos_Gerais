\begin{frame}[t]{Preparar e executar a prática}
\begin{enumerate}
\item Abrir \texttt{exemplos/pisca/pisca.ino} no Arduino IDE.
\item Selecionar Arduino Uno e o suporte de placas AVR correspondente.
\item Conectar o cabo USB de dados e selecionar a porta da placa.
\item Compilar; corrigir mensagens de erro antes do envio.
\item Enviar o programa e observar o LED integrado.
\item Alterar 500 para 250 nos dois intervalos e comparar o período.
\end{enumerate}
\small Compilar verifica o programa. Enviar grava a placa. A observação do LED verifica o comportamento físico.
\end{frame}
