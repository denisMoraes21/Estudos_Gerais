\begin{frame}[t]{Síntese: os oito encontros se conectam}
\begin{itemize}
\item Algoritmo e fluxograma definem o ciclo da aplicação.
\item Tipos e variáveis representam leituras e estados.
\item Operadores e condicionais expressam as decisões.
\item Repetições mantêm o monitoramento ativo.
\item Vetores e strings organizam conjuntos de dados.
\item Funções separam leitura, regra e atuação.
\item Testes verificam combinações, transições e resultados.
\item GPIO conecta a lógica à bancada.
\end{itemize}
\end{frame}
