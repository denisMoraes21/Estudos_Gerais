\begin{frame}[t]{Atividade prática e entrega}
\begin{enumerate}
\item Executar o exemplo \texttt{pisca}; medir o período aproximado.
\item Montar um botão e executar \texttt{botao\_led}.
\item Montar o segundo botão e executar \texttt{sinalizador}.
\item Preencher os quatro casos da tabela e testar a liberação de cada botão.
\item Explicar por que pressionado corresponde a \texttt{LOW}.
\end{enumerate}
\begin{block}{Entrega por dupla}
Código utilizado, esquema com nomes dos pinos, tabela esperado/observado e uma descrição de erro encontrado ou de verificação realizada.
\end{block}
\end{frame}
