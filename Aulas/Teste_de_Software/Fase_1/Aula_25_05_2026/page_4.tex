\begin{frame}{Plano de testes}
    Objetivo:
    \begin{itemize}
        \item Documenta os meios e o cronograma para atingir os objetivos;
        \item Ajuda a garantir que as atividades realizadas atenderão aos critérios; 
        \item Serve como um meio de comunicação com os membros da equipe e outros stakeholders; 
        \item Demonstra que os testes seguirão a política e a estratégia de testes existentes (ou explica por que os testes se desviarão delas);
    \end{itemize}
\end{frame}
