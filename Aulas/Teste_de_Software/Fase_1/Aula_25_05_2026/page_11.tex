\begin{frame}{Relatório de testes}
    Objetivo:
    \begin{itemize}
        \item Informar o status da qualidade do sistema
        \item Mostrar o progresso da execução dos testes.
        \item Registrar defeitos encontrados e seu impacto.
        \item Apoiar a decisão de release (go/no-go).
        \item Garantir transparência para stakeholders.
        \item Documentar evidências do processo de teste.
    \end{itemize}
\end{frame}
