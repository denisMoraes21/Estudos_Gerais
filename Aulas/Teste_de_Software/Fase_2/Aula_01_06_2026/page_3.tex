    \section*{Exemplo de Teste com unittest}

\begin{lstlisting}[float,caption={Código Python},label={lst:python}]
import unittest

def verificar_idade(idade):
    if idade >= 18:
        return "Maior de idade"
    else:
        return "Menor de idade"

assert verificar_idade(20) == "Maior de idade"
assert verificar_idade(15) == "Menor de idade"

class TestVerificarIdade(unittest.TestCase):

    def test_maior_de_idade(self):
        self.assertEqual(
            verificar_idade(20),
            "Maior de idade"
        )

    def test_menor_de_idade(self):
        self.assertEqual(
            verificar_idade(15),
            "Menor de idade"
        )
\end{lstlisting}
