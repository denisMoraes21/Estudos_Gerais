\begin{frame}{Demultiplexador -- DEMUX}

    O DEMUX realiza a operação conceitualmente oposta ao MUX.

    \[
        \boxed{\text{Uma entrada}}
        \longrightarrow
        \boxed{\text{DEMUX}}
        \longrightarrow
        \boxed{\text{Várias saídas}}
    \]

    \vspace{0.5cm}

    Os bits de seleção determinam para qual saída
    o sinal de entrada será enviado.

    \begin{alertblock}{Comparação}
        \[
            \text{MUX: várias entradas}
            \rightarrow
            \text{uma saída}
        \]

        \[
            \text{DEMUX: uma entrada}
            \rightarrow
            \text{várias saídas}
        \]
    \end{alertblock}

\end{frame}
