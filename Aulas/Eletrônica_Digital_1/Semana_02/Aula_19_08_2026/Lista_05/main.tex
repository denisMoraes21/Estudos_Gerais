Para cada função lógica:

\begin{enumerate}[label=\alph*)]
    \item Monte o Mapa de Karnaugh;
    \item Preencha corretamente as células;
    \item Faça os maiores agrupamentos possíveis;
    \item Determine a expressão lógica simplificada.
\end{enumerate}

\item \textbf{Mapa de Karnaugh com 2 variáveis}

Simplifique as funções utilizando mapas de Karnaugh com as variáveis
$A$ e $B$.

\begin{enumerate}[label=(\Alph*)]

    \item
    \[
        F(A,B)=\Sigma m(0,1)
    \]

    \item
    \[
        F(A,B)=\Sigma m(0,2)
    \]

    \item
    \[
        F(A,B)=\Sigma m(1,3)
    \]

    \item
    \[
        F(A,B)=\Sigma m(2,3)
    \]

    \item
    \[
        F(A,B)=\Sigma m(0,3)
    \]

    \item
    \[
        F(A,B)=\Sigma m(1,2)
    \]

    \item
    \[
        F(A,B)=\Sigma m(0,1,2)
    \]

    \item
    \[
        F(A,B)=\Sigma m(0,1,3)
    \]

    \item
    \[
        F(A,B)=\Sigma m(0,2,3)
    \]

    \item
    \[
        F(A,B)=\Sigma m(0,1,2,3)
    \]

\end{enumerate}

\vspace{0.5cm}

% =========================================================
% 3 VARIÁVEIS
% =========================================================

\item \textbf{Mapa de Karnaugh com 3 variáveis}

Simplifique as funções utilizando mapas de Karnaugh com as variáveis
$A$, $B$ e $C$.

\begin{enumerate}[label=(\Alph*)]

    \item
    \[
        F(A,B,C)=\Sigma m(0,1)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(2,3)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(0,2,4,6)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(1,3,5,7)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(0,1,2,3)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(4,5,6,7)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(0,1,4,5)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(2,3,6,7)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(0,1,2,3,5,7)
    \]

    \item
    \[
        F(A,B,C)=\Sigma m(1,2,3,5,7)
    \]

\end{enumerate}

\vspace{0.5cm}

% =========================================================
% 4 VARIÁVEIS
% =========================================================

\item \textbf{Mapa de Karnaugh com 4 variáveis}

Simplifique as funções utilizando mapas de Karnaugh com as variáveis
$A$, $B$, $C$ e $D$.

\begin{enumerate}[label=(\Alph*)]

    \item
    \[
        F(A,B,C,D)=\Sigma m(0,1)
    \]

    \item
    \[
        F(A,B,C,D)=\Sigma m(0,1,2,3)
    \]

    \item
    \[
        F(A,B,C,D)=\Sigma m(0,2,4,6)
    \]

    \item
    \[
        F(A,B,C,D)=\Sigma m(1,3,5,7)
    \]

    \item
    \[
    \begin{aligned}
        F(A,B,C,D) = \Sigma m(&0,1,4,5,\\
                              &8,9,12,13)
    \end{aligned}
    \]

    \item
    \[
    \begin{aligned}
        F(A,B,C,D) = \Sigma m(&2,3,6,7,\\
                              &10,11,14,15)
    \end{aligned}
    \]

    \item
    \[
    \begin{aligned}
        F(A,B,C,D) = \Sigma m(&0,1,2,3,\\
                              &8,9,10,11)
    \end{aligned}
    \]

    \item
    \[
    \begin{aligned}
        F(A,B,C,D) = \Sigma m(&4,5,6,7,\\
                              &12,13,14,15)
    \end{aligned}
    \]


    \item
    \[
    \begin{aligned}
        F(A,B,C,D) = \Sigma m(&0,2,3,5,7,8,\\
                              &10,11,13,15)
    \end{aligned}
    \]


    \item
    \[
    \begin{aligned}
        F(A,B,C,D) = \Sigma m(&0,2,3,5,7,8,\\
                              &10,11,13,15)
    \end{aligned}
    \]

\end{enumerate}

\end{enumerate}
