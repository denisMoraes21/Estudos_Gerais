    \small

    Analisando as quatro variáveis:

    \begin{center}
        \begin{tabular}{c|c|c}
            Variável & Comportamento & Resultado \\ \hline
            $A$ & varia & elimina \\
            $B$ & varia & elimina \\
            $C$ & permanece em $1$ & mantém $C$ \\
            $D$ & varia & elimina
        \end{tabular}
    \end{center}

    \vspace{0.5cm}

    Portanto:

    \[
        \cancel{A}\;
        \cancel{B}\;
        C\;
        \cancel{D}
    \]

    \begin{alertblock}{Expressão simplificada}
        \[
            \boxed{F=C}
        \]
    \end{alertblock}
