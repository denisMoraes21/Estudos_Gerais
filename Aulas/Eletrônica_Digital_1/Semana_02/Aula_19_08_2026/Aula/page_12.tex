\begin{frame}{Código ASCII}
    \small
    \begin{block}{Definição}
        ASCII significa
        \textit{American Standard Code for Information Interchange}.
    \end{block}
    O ASCII associa um valor numérico a cada caractere.
    Exemplos:
    \begin{center}
        \begin{tabular}{c|c|c|c}
            \textbf{Caractere} &
            \textbf{Decimal} &
            \textbf{Hexadecimal} &
            \textbf{Binário} \\ \hline

            A & 65 & 41 & 01000001 \\
            B & 66 & 42 & 01000010 \\
            a & 97 & 61 & 01100001 \\
            0 & 48 & 30 & 00110000 \\
            1 & 49 & 31 & 00110001
        \end{tabular}
    \end{center}

\end{frame}
