\begin{frame}{Mapa de Karnaugh: Analisando o grupo}
    \small

    O grupo utiliza todas as linhas AB = 00,\;01,\;11,\;10

    Portanto  A \text{ varia}, B \text{ varia}

    Logo \cancel{A}, \cancel{B}

    \begin{block}{Analisando as colunas}
        O grupo utiliza:

        \[
            CD=11
            \qquad\text{e}\qquad
            CD=10
        \]

        Comparando:

        \[
        \begin{array}{c|cc}
             & C & D \\ \hline
            11 & 1 & 1 \\
            10 & 1 & 0
        \end{array}
        \]

        $C$ permanece em $1$, enquanto $D$ varia.
    \end{block}

\end{frame}
