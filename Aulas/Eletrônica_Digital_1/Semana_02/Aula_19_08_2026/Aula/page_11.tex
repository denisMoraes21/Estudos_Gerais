\begin{frame}{Conversão Binário para Gray: Exemplo}
    \small
    Considere:
    \[
        B = 1011
    \]
    Primeiro bit:
    \[
        G_3 = 1
    \]
    Depois:
    \[
        G_2 = 1 \oplus 0 = 1
    \]
    \[
        G_1 = 0 \oplus 1 = 1
    \]
    \[
        G_0 = 1 \oplus 1 = 0
    \]
    Portanto:
    \[
        \boxed{1011_2 = 1110_{Gray}}
    \]
\end{frame}
