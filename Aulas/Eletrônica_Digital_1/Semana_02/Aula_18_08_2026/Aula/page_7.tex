\begin{frame}{Mapa de Karnaugh: Exemplo}
    \small

    Considere a seguinte função:

    \begin{center}
        \begin{tabular}{c c | c}
            $A$ & $B$ & $F$ \\ \hline
            0 & 0 & 0 \\
            0 & 1 & 1 \\
            1 & 0 & 0 \\
            1 & 1 & 1
        \end{tabular}
    \end{center}

    \begin{block}{Passo 1: Identificar onde $F=1$}
        A saída é $1$ para:

        \[
            AB=01
            \qquad\text{e}\qquad
            AB=11
        \]

        Portanto:

        \[
            F(A,B)=\Sigma m(1,3)
        \]
    \end{block}


\end{frame}
