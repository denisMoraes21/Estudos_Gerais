\begin{frame}{Representações de equações booleanas}

    \begin{itemize}
        \item \textbf{Método algébrico}
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Exemplo: Simplifique a expressão}

    \[
        F = AB + A\overline{B} + BC
    \]

    \textbf{1. Agrupando os dois primeiros termos:}

    \[
        F = (AB + A\overline{B}) + BC
    \]

    \textbf{2. Colocando $A$ em evidência:}

    \[
        F = A(B + \overline{B}) + BC
    \]

\end{frame}
