\begin{frame}{Conversões}
    \begin{itemize}
        \item Conversão decimal/hexadecimal
    \end{itemize}
    Exemplo: 245 \\[0.2cm]
    \centering
    $245 \div 16 = 15$ resto 5 \\[0.2cm]
    $15 \div 16 = 0$ resto 15 = F \\[0.4cm]
    \vspace{0.2cm}

    $245_{10} = F5_{16}$
\end{frame}
