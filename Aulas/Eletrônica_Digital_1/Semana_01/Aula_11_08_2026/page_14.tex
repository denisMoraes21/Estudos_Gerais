\begin{frame}{Conversões}
    \begin{itemize}
        \item Conversão octal/hexadecimal
    \end{itemize}
    Não existe uma correspondência. \\[0.2cm]
    Octal - Binário - Hexadecimal
    \vspace{0.2cm}
    Exemplo: 365 \\[0.2cm]
    \centering
    3 = 011, 6 = 110, 5 = 101 \\[0.2cm]
    $365_{8}$ = $011110101_{2}$ \\[0.2cm]
    0000 1111 0101 \\[0.2cm]
    0000 = 0, 1111 = F, 0101 = 5 \\[0.2cm]
    $365_{8}$ = $F5_{16}$
\end{frame}
