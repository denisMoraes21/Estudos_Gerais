\begin{frame}{Bases numéricas}
    \begin{itemize}
        \item Decimal (10 Símbolos)
    \end{itemize}

    \vspace{0.5cm}

    Exemplo:

    \begin{center}
        $245_{10}$
    \end{center}

    \[
        245 = 2 \times 10^{2}
        + 4 \times 10^{1}
        + 5 \times 10^{0}
        = 200 + 40 + 5
    \]

    \vspace{0.5cm}

    \textbf{Tente você:} $35865_{10}$
\end{frame}
