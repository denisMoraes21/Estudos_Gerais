\begin{frame}{Bases numéricas}
    \begin{itemize}
        \item Octal (8 Símbolos)
    \end{itemize}

    \vspace{0.5cm}

    Exemplo:

    \begin{center}
        $245_{8}$
    \end{center}

    \[
        245_{8}
        = 2 \times 8^{2}
        + 4 \times 8^{1}
        + 5 \times 8^{0}
        = 128 + 32 + 5
        = 165_{10}
    \]

    \vspace{0.5cm}

    \textbf{Tente você:} $35865_{8}$
\end{frame}
