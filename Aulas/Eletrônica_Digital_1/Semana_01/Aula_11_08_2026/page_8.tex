\begin{frame}{Conversões}
    \begin{itemize}
        \item Conversão decimal/binário
    \end{itemize}
    Exemplo: 45 \\[0.2cm]
    \centering
    $45 \div 2 = 22$ resto 1 \\[0.2cm]
    $22 \div 2 = 11$ resto 0 \\[0.2cm]
    $11 \div 2 = 5$ resto 1 \\[0.2cm]
    $5 \div 2 = 2$ resto 1 \\[0.2cm]
    $2 \div 2 = 1$ resto 0 \\[0.2cm]
    $1 \div 2 = 0$ resto 1 \\[0.4cm]
    \vspace{0.2cm}

    $45_{10} = 101101_{2}$
\end{frame}
