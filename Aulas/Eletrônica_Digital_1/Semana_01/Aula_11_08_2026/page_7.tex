\begin{frame}{Bases numéricas}
    \begin{itemize}
        \item Hexadecimal (16 Símbolos)
    \end{itemize}

    \vspace{0.5cm}

    Exemplo:

    \begin{center}
        $2F_{16}$
    \end{center}

    \[
        2F_{16}
        = 2 \times 16^{1}
        + 15 \times 16^{0}
        = 32 + 15
        = 47_{10}
    \]

    \vspace{0.5cm}

    \textbf{Tente você:} $3A_{16}$
\end{frame}
