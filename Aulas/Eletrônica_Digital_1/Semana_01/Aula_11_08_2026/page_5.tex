\begin{frame}{Bases numéricas}
    \begin{itemize}
        \item Binário (2 Símbolos)
    \end{itemize}

    \vspace{0.5cm}

    Exemplo:

    \begin{center}
        $1011_{2}$
    \end{center}

    \[
        1011_{2}
        = 1 \times 2^{3}
        + 0 \times 2^{2}
        + 1 \times 2^{1}
        + 1 \times 2^{0}
        = 8 + 0 + 2 + 1
        = 11_{10}
    \]

    \vspace{0.5cm}

    \textbf{Tente você:} $100011_{2}$

\end{frame}
