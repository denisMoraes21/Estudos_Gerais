% =========================================================
% SOMA
% =========================================================

\item Qual é o resultado da soma
$1010_2 + 0011_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $1011_2$
    \item $1100_2$
    \item $1101_2$
    \item $1110_2$
    \item $1111_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o resultado da soma
$1111_2 + 0001_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $1111_2$
    \item $10000_2$
    \item $10001_2$
    \item $11000_2$
    \item $11100_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o resultado da soma
$1011_2 + 0110_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $01111_2$
    \item $10000_2$
    \item $10001_2$
    \item $10010_2$
    \item $10101_2$
\end{enumerate}

\vspace{0.2cm}


% =========================================================
% SUBTRAÇÃO
% =========================================================

\item Qual é o resultado da subtração
$1010_2 - 0011_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $0101_2$
    \item $0110_2$
    \item $0111_2$
    \item $1000_2$
    \item $1001_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o resultado da subtração
$1100_2 - 0101_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $0101_2$
    \item $0110_2$
    \item $0111_2$
    \item $1000_2$
    \item $1001_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o resultado da subtração
$1111_2 - 0110_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $0111_2$
    \item $1000_2$
    \item $1001_2$
    \item $1010_2$
    \item $1011_2$
\end{enumerate}

\vspace{0.2cm}


% =========================================================
% MULTIPLICAÇÃO
% =========================================================

\item Qual é o resultado da multiplicação
$101_2 \times 11_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $1010_2$
    \item $1100_2$
    \item $1101_2$
    \item $1111_2$
    \item $10000_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o resultado da multiplicação
$110_2 \times 10_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $0110_2$
    \item $1000_2$
    \item $1010_2$
    \item $1100_2$
    \item $1110_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o resultado da multiplicação
$111_2 \times 11_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $10001_2$
    \item $10010_2$
    \item $10100_2$
    \item $10101_2$
    \item $11001_2$
\end{enumerate}

\vspace{0.2cm}


% =========================================================
% DIVISÃO
% =========================================================

\item Qual é o resultado da divisão
$1100_2 \div 10_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $10_2$
    \item $100_2$
    \item $101_2$
    \item $110_2$
    \item $111_2$
\end{enumerate}

\vspace{0.2cm}


\item Qual é o quociente da divisão
$1111_2 \div 11_2$?

\begin{enumerate}[label=(\Alph*)]
    \item $10_2$
    \item $11_2$
    \item $100_2$
    \item $101_2$
    \item $110_2$
\end{enumerate}

\vspace{0.2cm}


\item Na divisão
$1101_2 \div 10_2$,
qual é o quociente e o resto?

\begin{enumerate}[label=(\Alph*)]
    \item Quociente $101_2$, resto $1_2$
    \item Quociente $110_2$, resto $0_2$
    \item Quociente $110_2$, resto $1_2$
    \item Quociente $111_2$, resto $0_2$
    \item Quociente $111_2$, resto $1_2$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% NOT
% =========================================================

\item Qual é o resultado de
$\mathrm{NOT}(10110010_2)$?

\begin{enumerate}[label=(\Alph*)]
    \item $01001101_2$
    \item $01001110_2$
    \item $10110001_2$
    \item $11001101_2$
    \item $01101101_2$
\end{enumerate}

\vspace{0.2cm}


\item Considerando 8 bits, qual é o resultado de
$\mathrm{NOT}(11110000_2)$?

\begin{enumerate}[label=(\Alph*)]
    \item $00001111_2$
    \item $00011111_2$
    \item $00110011_2$
    \item $11110000_2$
    \item $11111111_2$
\end{enumerate}

\vspace{0.2cm}


% =========================================================

    \item $1110_2$
