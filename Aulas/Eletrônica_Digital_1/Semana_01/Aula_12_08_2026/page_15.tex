\begin{frame}{Deslocamento de bits para esquerda (Shift Left)}
    \begin{itemize}
        \item Desloca os bits para a esquerda.
        \item As posições vazias à direita são preenchidas com $0$.
    \end{itemize}

    \[
    \begin{array}{rcl}
        0011_{2} << 1 & = & 0110_{2} \\
        0011_{2} << 2 & = & 1100_{2}
    \end{array}
    \]

    \vspace{0.3cm}

    \[
        x << n = x \times 2^n
    \]

\end{frame}
