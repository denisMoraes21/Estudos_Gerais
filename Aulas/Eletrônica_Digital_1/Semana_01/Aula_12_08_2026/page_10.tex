\begin{frame}{Divisão entre números binários} 
    \begin{itemize} 
        \item Divisão com dois números: 
    \end{itemize} \[ 
    \begin{array}{r|l} 1100_{2} & 10_{2} \\ \cline{2-2} -10 & 110_{2} \\ \phantom{-}10 & \\ -10 & \\ \phantom{-}00 & \\ \end{array} \] 
    \begin{itemize}
        \item Divisão com resto:
    \end{itemize} \[
    \begin{array}{r|l}
        1101_{2} & 10_{2} \\
        \cline{2-2}
        -10      & 110_{2} \\
        \phantom{-}10 & \\
        -10      & \\
        \phantom{-}01 & \\
    \end{array}
    \]

    \[
        1101_{2} \div 10_{2}
        = 110_{2}
        \quad \text{com resto } 1_{2}
    \]
\end{frame}
