\begin{frame}{Identidades lógicas: Lei da Identidade}
    \begin{block}{Definição}
        \begin{itemize}
            \item é realizada uma operação \textbf{OR} com $0$;
            \item é realizada uma operação \textbf{AND} com $1$.
        \end{itemize}
    \end{block}

    \vspace{0.1cm}

    \begin{columns}[T]

        % Coluna OR
        \begin{column}{0.47\textwidth}
            \begin{block}{Identidade OR}
                \[
                    \boxed{A + 0 = A}
                \]
                \[
                    0 + 0 = 0
                \]
                \[
                    1 + 0 = 1
                \]

                Portanto, o valor de $A$ não é alterado.
            \end{block}
        \end{column}

        % Coluna AND
        \begin{column}{0.47\textwidth}
            \begin{block}{Identidade AND}
                \[
                    \boxed{A \cdot 1 = A}
                \]
                \[
                    0 \cdot 1 = 0
                \]
                \[
                    1 \cdot 1 = 1
                \]

                Portanto, o valor de $A$ não é alterado.
            \end{block}
        \end{column}

    \end{columns}

    \begin{alertblock}{Interpretação}
        O elemento $0$ é a identidade da operação \textbf{OR}, enquanto
        o elemento $1$ é a identidade da operação \textbf{AND}.
    \end{alertblock}

    
\end{frame}
