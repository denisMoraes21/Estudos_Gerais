\begin{frame}{Identidades lógicas: Lei da Dominação}
    \small

    \begin{block}{Definição}
        Na operação OR, o valor lógico $1$ domina o resultado.
        Na operação AND, o valor lógico $0$ domina o resultado.
    \end{block}

    \begin{columns}[T]

        % Primeira forma
        \small
        \begin{column}{0.49\textwidth}
            \begin{block}{Dominação com OR}
                \[
                    \boxed{A + 1 = 1}
                \]
                \[
                    A = 0
                    \quad\Rightarrow\quad
                    0 + 1 = 1
                \]
                \[
                    A = 1
                    \quad\Rightarrow\quad
                    1 + 1 = 1
                \]
                \[
                    \boxed{A+1=1}
                \]
            \end{block}
        \end{column}

        % Segunda forma
        \small
        \begin{column}{0.49\textwidth}
            \begin{block}{Dominação com AND}
                \[
                    \boxed{A\cdot0=0}
                \]
                \[
                    A = 0
                    \quad\Rightarrow\quad
                    0\cdot0=0
                \]
                \[
                    A = 1
                    \quad\Rightarrow\quad
                    1\cdot0=0
                \]
                \[
                    \boxed{A\cdot0=0}
                \]
            \end{block}
        \end{column}

    \end{columns}

\end{frame}
