\begin{frame}{Identidades lógicas: Lei associativa}
    \begin{block}{Definição}
        A forma como as variáveis são agrupadas não altera o resultado
        das operações \textbf{OR} e \textbf{AND}.
    \end{block}
    \begin{columns}[T]

        % Coluna OR
        \begin{column}{0.5\textwidth}
            \begin{block}{Associativa OR}
                \[
                    \boxed{(A + B) + C = A + (B + C)}
                \]
                \[
                    (1 + 0) + 1
                \]
                \[
                    = 1 + (0 + 1)
                \]
                \[
                    = 1
                \]
            \end{block}
        \end{column}

        % Coluna AND
        \begin{column}{0.5\textwidth}
            \begin{block}{Associativa AND}
                \[
                    \boxed{(A \cdot B)\cdot C
                    = A\cdot(B\cdot C)}
                \]
                \[
                    (1 \cdot 1)\cdot 0
                \]
                \[
                    = 1\cdot(1\cdot0)
                \]
                \[
                    = 0
                \]
            \end{block}
        \end{column}

    \end{columns}
    
\end{frame}
