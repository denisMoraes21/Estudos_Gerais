\begin{frame}{Identidades lógicas: Lei comutativa}
    \begin{block}{Definição}
        A ordem das variáveis não altera o resultado das operações
        \textbf{OR} e \textbf{AND}.
    \end{block}

    \begin{columns}[T]

        % Coluna OR
        \begin{column}{0.5\textwidth}
            \begin{block}{Comutativa OR}
                \[
                    \boxed{A + B = B + A}
                \]
                \[
                    0 + 1 = 1 + 0 = 1
                \]
                \[
                    1 + 1 = 1 + 1 = 1
                \]

                A ordem das entradas não altera
                a saída da porta OR.
            \end{block}
        \end{column}

        % Coluna AND
        \begin{column}{0.5\textwidth}
            \begin{block}{Comutativa AND}
                \[
                    \boxed{A \cdot B = B \cdot A}
                \]
                \[
                    0 \cdot 1 = 1 \cdot 0 = 0
                \]
                \[
                    1 \cdot 1 = 1 \cdot 1 = 1
                \]

                A ordem das entradas não altera
                a saída da porta AND.
            \end{block}
        \end{column}

    \end{columns}
    
\end{frame}
