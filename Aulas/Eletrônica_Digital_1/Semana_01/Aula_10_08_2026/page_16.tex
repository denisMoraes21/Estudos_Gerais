\begin{frame}{Faixas de tensão entre tecnologias}
    Família CMOS (Complementary Metal-Oxide-Semiconductor):
    \begin{itemize}
        \item De 0,0V até 1,5V - 0 lógico
        \item De 5,0V até 6,5V - 1 lógico
        \item Maior imunidade a ruídos
    \end{itemize}
    Família TTL (Transistor-Transistor Logic):
    \begin{itemize}
        \item De 0,0V até 0,8V - 0 lógico
        \item De 2,0V até 5,0V - 1 lógico
        \item Menor imunidade a ruídos
    \end{itemize}
\end{frame}
