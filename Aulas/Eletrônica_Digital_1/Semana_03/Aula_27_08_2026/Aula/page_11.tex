
\begin{frame}{Principais elementos da lógica sequencial}

    \begin{block}{Elementos básicos}
        \begin{itemize}
            \item \textbf{Latch:} armazena um bit e é sensível ao nível
            de um sinal de controle
            \item \textbf{Flip-flop:} armazena um bit e normalmente é
            acionado pela borda do clock
            \item \textbf{Registrador:} conjunto de flip-flops utilizado
            para armazenar vários bits
            \item \textbf{Contador:} circuito que percorre uma sequência
            de estados
        \end{itemize}
    \end{block}

    \begin{alertblock}{Base dos sistemas digitais}
        Esses elementos são utilizados em processadores, controladores,
        memórias, temporizadores e sistemas de automação.
    \end{alertblock}

\end{frame}
