\begin{frame}{Sinal de clock}

    \begin{block}{Definição}
        O clock é um sinal periódico utilizado para sincronizar as
        operações de um circuito sequencial.
    \end{block}

    \begin{columns}[T]

        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Nível lógico alto}
                \[
                    CLK = 1
                \]

                O sinal está em seu nível de tensão alto.
            \end{alertblock}
        \end{column}

        \begin{column}{0.47\textwidth}
            \begin{block}{Nível lógico baixo}
                \[
                    CLK = 0
                \]

                O sinal está em seu nível de tensão baixo.
            \end{block}
        \end{column}

    \end{columns}

\end{frame}
