\begin{frame}{Estado atual e próximo estado}

    \begin{block}{Estado atual}
        O estado atual, representado por $Q(t)$, corresponde à informação
        armazenada no instante presente.
    \end{block}

    \begin{block}{Próximo estado}
        O próximo estado, representado por $Q(t+1)$, corresponde ao valor
        que será armazenado após um evento de atualização.
    \end{block}

    \begin{alertblock}{Relação geral}
        \[
            Q(t+1) = f\bigl(Q(t),X(t)\bigr)
        \]

        O próximo estado depende do estado atual $Q(t)$ e das entradas
        atuais $X(t)$.
    \end{alertblock}

\end{frame}
