\begin{frame}{Memória em circuitos digitais}

    \begin{block}{Elemento de memória}
        Um elemento de memória é um circuito capaz de armazenar
        temporariamente um valor binário.
    \end{block}

    \begin{columns}[T]

        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Valor armazenado}
                \begin{itemize}
                    \item Estado lógico 0
                    \item Estado lógico 1
                    \item Um bit de informação
                \end{itemize}
            \end{alertblock}
        \end{column}

        \begin{column}{0.47\textwidth}
            \begin{block}{Elementos utilizados}
                \begin{itemize}
                    \item Latches
                    \item Flip-flops
                    \item Registradores
                    \item Memórias
                \end{itemize}
            \end{block}
        \end{column}

    \end{columns}

\end{frame}
