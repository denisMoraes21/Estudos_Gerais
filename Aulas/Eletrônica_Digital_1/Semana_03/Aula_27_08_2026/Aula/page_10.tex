\begin{frame}{Circuitos síncronos e assíncronos}

    \begin{columns}[T]

        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Circuitos síncronos}
                \begin{itemize}
                    \item Utilizam um sinal de clock
                    \item Os estados mudam em instantes definidos
                    \item Possuem operação coordenada
                    \item São mais fáceis de projetar
                \end{itemize}
            \end{alertblock}
        \end{column}

        \begin{column}{0.47\textwidth}
            \begin{block}{Circuitos assíncronos}
                \begin{itemize}
                    \item Não dependem de um clock global
                    \item Os estados mudam com as entradas
                    \item Podem responder mais rapidamente
                    \item Exigem maior cuidado no projeto
                \end{itemize}
            \end{block}
        \end{column}

    \end{columns}

\end{frame}
