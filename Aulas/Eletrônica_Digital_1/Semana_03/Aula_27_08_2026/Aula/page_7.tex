
\begin{frame}{Realimentação}

    \begin{block}{Definição}
        A realimentação ocorre quando uma saída do circuito retorna
        para uma de suas entradas.
    \end{block}

    \begin{alertblock}{Função da realimentação}
        A realimentação permite que o circuito mantenha uma informação
        mesmo após a entrada responsável pela mudança ser removida.
    \end{alertblock}

    \begin{block}{Consequência}
        Diferentemente dos circuitos combinacionais, a saída pode
        permanecer em seu estado anterior.
    \end{block}

\end{frame}
