\begin{frame}{O que é uma família lógica?}
    \begin{block}{Definição}
        Uma família lógica é um conjunto de circuitos integrados construídos
        com uma mesma tecnologia e com características elétricas compatíveis.
    \end{block}

    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{TTL}
                \begin{itemize}
                    \item Baseada em transistores bipolares (BJT)
                    \item Tradicionalmente alimentada com 5~V
                    \item Exemplos: 74, 74LS, 74F e 74AS
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{CMOS}
                \begin{itemize}
                    \item Baseada em transistores MOSFET
                    \item Ampla variedade de tensões e velocidades
                    \item Exemplos: CD4000, HC, HCT, LVC e ALVC
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
