\begin{enumerate}[leftmargin=*]

% =========================================================
% QUESTÃO 1
% =========================================================

\item \textbf{(0,5 pontos)} Utilizando as propriedades da Álgebra Booleana,
qual é a forma simplificada da expressão

\[
F=A+AB?
\]

\begin{enumerate}[label=(\Alph*)]
    \item $F=AB$
    \item $F=A$
    \item $F=B$
    \item $F=A+B$
    \item $F=1$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 2
% =========================================================

\item \textbf{(0,5 pontos)} Simplifique algebricamente a expressão:

\[
F=AB+A\overline{B}
\]

\begin{enumerate}[label=(\Alph*)]
    \item $F=A$
    \item $F=B$
    \item $F=\overline{A}$
    \item $F=A+B$
    \item $F=AB$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 3
% =========================================================

\item \textbf{(0,5 pontos)} Qual propriedade da Álgebra Booleana permite
realizar a simplificação

\[
A+A B=A?
\]

\begin{enumerate}[label=(\Alph*)]
    \item Lei comutativa
    \item Lei associativa
    \item Lei da absorção
    \item Lei de De Morgan
    \item Lei do complemento
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 4
% =========================================================

\item \textbf{(0,5 pontos)} Aplicando a Lei de De Morgan, qual expressão é
equivalente a

\[
\overline{A+B}?
\]

\begin{enumerate}[label=(\Alph*)]
    \item $\overline{A}+\overline{B}$
    \item $\overline{A}\cdot\overline{B}$
    \item $A\cdot B$
    \item $\overline{A}\cdot B$
    \item $A+\overline{B}$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 5
% =========================================================

\item \textbf{(0,5 pontos)} Considere o mapa de Karnaugh de duas variáveis:

\[
\begin{array}{c|cc}
A\backslash B & 0 & 1 \\ \hline
0 & 0 & 1 \\
1 & 1 & 1
\end{array}
\]

Qual é a expressão simplificada representada pelo mapa?

\begin{enumerate}[label=(\Alph*)]
    \item $F=AB$
    \item $F=A+B$
    \item $F=\overline{A}+\overline{B}$
    \item $F=A\oplus B$
    \item $F=\overline{A}B$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 6
% =========================================================

\item \textbf{(0,5 pontos)} Uma função de três variáveis é representada por

\[
F(A,B,C)=\Sigma m(1,3,5,7).
\]

Ao agrupar os valores no mapa de Karnaugh, qual é a expressão simplificada?

\begin{enumerate}[label=(\Alph*)]
    \item $F=A$
    \item $F=B$
    \item $F=C$
    \item $F=\overline{C}$
    \item $F=A+B+C$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 7
% =========================================================

\item \textbf{(0,5 pontos)} Considere a função

\[
F(A,B,C)=\Sigma m(0,2,4,6).
\]

Qual é o resultado da simplificação utilizando o mapa de Karnaugh?

\begin{enumerate}[label=(\Alph*)]
    \item $F=C$
    \item $F=\overline{C}$
    \item $F=A$
    \item $F=\overline{B}$
    \item $F=A+B$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 8
% =========================================================

\item \textbf{(0,5 pontos)} Considere a função de quatro variáveis:

\[
F(A,B,C,D)=\Sigma m(0,2,8,10).
\]

Qual expressão é obtida após formar o maior agrupamento possível no mapa
de Karnaugh?

\begin{enumerate}[label=(\Alph*)]
    \item $F=\overline{A}\,\overline{B}$
    \item $F=\overline{B}\,\overline{D}$
    \item $F=\overline{C}\,\overline{D}$
    \item $F=A\overline{D}$
    \item $F=\overline{B}+D$
\end{enumerate}

\vspace{0.2cm}

% =========================================================
% QUESTÃO 9
% =========================================================

\item \textbf{(0,5 pontos)} Considere a função

\[
F(A,B,C,D)=
\Sigma m(0,2,5,7,8,10,13,15).
\]

Qual é a expressão simplificada?

\begin{enumerate}[label=(\Alph*)]
    \item $F=\overline{B}\,\overline{D}+BD$
    \item $F=\overline{A}C+A\overline{C}$
    \item $F=B+D$
    \item $F=\overline{B}D+B\overline{D}$
    \item $F=A+C$
\end{enumerate}

\vspace{0.2cm}

    \item $S=\overline{A}B+AC$
    \item $S=A\overline{B}+AC$
    \item $S=\overline{A}B+\overline{A}C$
    \item $S=\overline{A+B}+AC$
\end{enumerate}

\end{enumerate}
