\begin{frame}[fragile,t]{Contador: implementação em VHDL}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{7}{8}\selectfont]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;
entity contador4 is
  port (clk, rst, en : in std_logic;
        q : out std_logic_vector(3 downto 0));
end entity;
architecture rtl of contador4 is
  signal cont : unsigned(3 downto 0);
begin
  process(clk)
  begin
    if rising_edge(clk) then
      if rst = '1' then cont <= (others => '0');
      elsif en = '1' then cont <= cont + 1;
      end if;
    end if;
  end process;
  q <= std_logic_vector(cont);
end architecture;
\end{lstlisting}
