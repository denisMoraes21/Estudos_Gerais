\begin{frame}[t]{Aplicações em sistemas digitais}
  \begin{itemize}
    \item \textbf{Computadores:} RAM para programas em execução; SSD para arquivos.
    \item \textbf{Microcontroladores:} Flash para firmware; RAM para variáveis e pilha.
    \item \textbf{IoT:} EEPROM/Flash para credenciais, configuração de rede e calibração.
    \item \textbf{Sistemas embarcados:} memória não volátil para registrar eventos e falhas.
    \item \textbf{FPGA:} memória de configuração e blocos internos de RAM.
  \end{itemize}
\end{frame}
