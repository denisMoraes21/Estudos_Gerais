\begin{frame}[t]{EEPROM --- memória apagável eletricamente}
  \begin{block}{EEPROM}
    \textit{Electrically Erasable Programmable Read-Only Memory}: memória não volátil que pode ser apagada e regravada eletricamente.
  \end{block}
  \begin{itemize}
    \item Permite alterar pequenas quantidades de dados.
    \item Geralmente permite apagar/escrever por bytes ou páginas pequenas.
    \item Possui número limitado de ciclos de escrita.
  \end{itemize}
  \begin{exampleblock}{Aplicações}
    Calibração, número de série, senha, parâmetros de rede e configurações do usuário.
  \end{exampleblock}
\end{frame}
