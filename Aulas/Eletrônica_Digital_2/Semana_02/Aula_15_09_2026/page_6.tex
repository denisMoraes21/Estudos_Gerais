\begin{frame}[t]{RAM --- Random Access Memory}
  \begin{block}{Características}
    Memória de leitura e escrita, normalmente \textbf{volátil}, com acesso direto a qualquer endereço.
  \end{block}
  \begin{itemize}
    \item Armazena variáveis, pilha (stack), heap e buffers.
    \item É rápida e usada enquanto o sistema está ligado.
    \item Ao desligar o equipamento, seu conteúdo é perdido.
  \end{itemize}
  \begin{exampleblock}{Em um microcontrolador}
    Uma variável \texttt{contador} e um buffer de recepção UART ficam, em geral, na RAM.
  \end{exampleblock}
\end{frame}
