\begin{frame}[t]{Resumo}
  \begin{itemize}
    \item \textbf{RAM:} volátil, rápida, leitura e escrita temporária.
    \item \textbf{ROM:} não volátil, conteúdo predominantemente fixo.
    \item \textbf{EEPROM:} não volátil, adequada para pequenos parâmetros regraváveis.
    \item \textbf{Flash:} não volátil, alta capacidade, apagamento por blocos/setores.
    \item A organização usa endereços, palavras e sinais de controle.
  \end{itemize}
\end{frame}
