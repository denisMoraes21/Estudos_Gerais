\begin{frame}[t]{O que é uma memória digital?}
  \begin{block}{Definição}
    Memória é um circuito capaz de \textbf{armazenar informação binária} para leitura posterior.
  \end{block}
  \begin{center}
    \fbox{\parbox{0.90\textwidth}{\centering\small
      \textbf{Dados de entrada} $\xrightarrow{\text{escrita}}$
      \textbf{Memória} $\xrightarrow{\text{leitura}}$
      \textbf{Dados de saída}}}
  \end{center}
  \begin{exampleblock}{Exemplos cotidianos}
    Configurações de um microcontrolador, arquivos em um pendrive e variáveis de um programa em execução.
  \end{exampleblock}
\end{frame}
