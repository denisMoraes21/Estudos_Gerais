\begin{frame}[t]{Memória volátil e não volátil}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Volátil}
      Perde os dados quando a alimentação é removida.
      \begin{itemize}
        \item Exemplo: RAM.
        \item Uso: dados temporários durante a execução.
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Não volátil}
      Mantém os dados sem alimentação.
      \begin{itemize}
        \item Exemplos: ROM, EEPROM e Flash.
        \item Uso: programa, configuração e arquivos.
      \end{itemize}
    \end{block}
  \end{columns}
  \begin{alertblock}{Ideia-chave}
    Não volátil não significa necessariamente que pode ser escrita infinitamente.
  \end{alertblock}
\end{frame}
