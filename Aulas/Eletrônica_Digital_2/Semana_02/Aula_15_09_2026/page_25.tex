\begin{frame}{Aplicações}
  \begin{itemize}
    \item \textbf{Conversores ADC:} estabilização da tensão durante a conversão.
    \item \textbf{Sistemas de aquisição de dados (DAQ):} leitura confiável de sensores.
    \item \textbf{Áudio digital:} captura periódica do sinal antes da conversão.
    \item \textbf{Instrumentação:} medição de sinais que variam rapidamente.
    \item \textbf{Multiplexação de canais:} retenção do valor de um canal enquanto outro é selecionado.
    \item \textbf{Osciloscópios digitais:} captura de amostras de um sinal para processamento e exibição.
  \end{itemize}
\end{frame}
