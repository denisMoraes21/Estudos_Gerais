\begin{frame}[t]{Tipos comuns de RAM}
  \small
  \textbf{SRAM}: \textit{Static Random Access Memory}\\
  Memória estática de acesso aleatório.\\[0.15cm]
  \textbf{DRAM}: \textit{Dynamic Random Access Memory}\\
  Memória dinâmica de acesso aleatório.
  \vspace{0.25cm}

  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{>{\raggedright\arraybackslash}p{0.22\textwidth} p{0.34\textwidth} p{0.34\textwidth}}
    \toprule
    \textbf{Tipo} & \textbf{Característica} & \textbf{Uso típico} \\
    \midrule
    SRAM & Sem atualização periódica; rápida e mais cara. & Cache e RAM de microcontroladores. \\
    DRAM & Exige refresh; alta densidade e menor custo. & Memória principal de computadores. \\
    \bottomrule
  \end{tabular}
  \vspace{0.15cm}
  \begin{alertblock}{Atenção}
    Ambas são voláteis: perdem os dados sem alimentação.
  \end{alertblock}
\end{frame}
