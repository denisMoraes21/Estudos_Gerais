\begin{frame}{I\textsuperscript{2}C e conceitos digitais}
\begin{itemize}
  \item Comunicação \textbf{síncrona}: \(SCL\) define o instante dos bits em \(SDA\).
  \item Comunicação por \textbf{barramento}: vários circuitos compartilham as mesmas linhas.
  \item \textbf{Endereçamento}: cada periférico recebe um endereço no protocolo.
  \item \textbf{Registradores}: cada lado envia e recebe bytes serialmente.
\end{itemize}
\end{frame}
