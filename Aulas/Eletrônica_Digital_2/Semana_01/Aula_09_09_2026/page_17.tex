\begin{frame}{I\textsuperscript{2}C}
\begin{itemize}
  \item \textbf{I\textsuperscript{2}C} usa duas linhas: \(SCL\), clock, e \(SDA\), dados.
  \item Um mestre inicia a comunicação e envia o endereço do dispositivo desejado.
  \item Vários dispositivos podem usar as mesmas duas linhas.
  \item Cada byte transferido recebe uma resposta \textit{ACK} ou \textit{NACK}.
\end{itemize}
\begin{block}{Aplicações}
Sensores de temperatura, RTC, memória EEPROM e conversores de sinais.
\end{block}
\end{frame}
