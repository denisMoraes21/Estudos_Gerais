\begin{frame}{Tipos de registradores}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l|l}
Tipo & Entrada & Saída \\
\hline
SISO & Serial & Serial \\
SIPO & Serial & Paralela \\
PISO & Paralela & Serial \\
PIPO & Paralela & Paralela \\
\end{tabular}
\end{center}

\vspace{0.3cm}
\begin{itemize}
  \item \textbf{S} significa \textit{serial}; \textbf{P} significa \textit{paralela}.
  \item O primeiro termo indica a entrada; o segundo indica a saída.
\end{itemize}
\end{frame}
