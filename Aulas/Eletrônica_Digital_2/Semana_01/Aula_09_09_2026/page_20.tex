\begin{frame}{Estados estáveis}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Estado estável}
\begin{itemize}
\item O circuito permanece nesse estado sem ação externa.
\item A saída mantém nível lógico 0 ou 1.
\end{itemize}
\column{0.48\textwidth}
\textbf{Estado instável}
\begin{itemize}
\item O circuito sai desse estado após um intervalo de tempo.
\item Resistores e capacitores podem definir esse intervalo.
\end{itemize}
\end{columns}
\end{frame}
