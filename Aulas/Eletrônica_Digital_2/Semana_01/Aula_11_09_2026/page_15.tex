\begin{frame}{Equação de saída do DAC}
    Para um DAC de $n$ bits:

    \[
        V_{out} =
        \frac{\text{valor digital}}{2^n-1}
        \times V_{ref}
    \]

    \begin{exampleblock}{Exemplo: DAC de 8 bits}
        Considere:

        \[
            V_{ref} = 3{,}3 \text{ V}
            \qquad
            \text{entrada} = 128
        \]

        \[
            V_{out} =
            \frac{128}{255} \times 3{,}3
            \approx 1{,}66 \text{ V}
        \]
    \end{exampleblock}
\end{frame}
