\begin{frame}{Aplicações do DAC}
    \begin{itemize}
        \item geração de áudio;
        \item controle de brilho;
        \item geração de sinais;
        \item controle de motores;
        \item ajuste de referência de tensão;
        \item instrumentos e sistemas de automação.
    \end{itemize}

    \begin{exampleblock}{Exemplo prático}
        Um sistema digital armazena uma música como números. O DAC transforma
        esses números em um sinal elétrico que pode ser amplificado e enviado
        para um alto-falante.
    \end{exampleblock}
\end{frame}
