\begin{frame}{Parâmetros importantes}
  \begin{tabular}{p{0.30\textwidth} p{0.62\textwidth}}
    \textbf{Tempo de aquisição} & Tempo necessário para o capacitor atingir uma tensão suficientemente próxima da entrada. \\[0.18cm]
    \textbf{Tempo de abertura} & Tempo entre o comando de controle e o desligamento efetivo da chave. \\[0.18cm]
    \textbf{Erro de retenção} & Diferença entre a tensão armazenada e a tensão ideal desejada. \\[0.18cm]
    \textbf{Droop rate} & Queda lenta da tensão durante o modo \textit{hold}, causada por correntes de fuga. \\[0.18cm]
    \textbf{Abertura de banda} & Limita quão rapidamente o circuito consegue acompanhar variações da entrada no modo \textit{track}. \\
  \end{tabular}
\end{frame}
