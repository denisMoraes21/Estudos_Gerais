\begin{frame}{Etapas da conversão ADC}
    \begin{enumerate}
        \item \textbf{Amostragem:} mede o sinal em instantes definidos.
        \item \textbf{Quantização:} associa a amostra a um nível digital.
        \item \textbf{Codificação:} representa esse nível em bits.
    \end{enumerate}
