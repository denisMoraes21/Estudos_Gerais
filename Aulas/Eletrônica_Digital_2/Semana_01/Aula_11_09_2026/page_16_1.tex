\begin{frame}{PWM como saída analógica aproximada}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \item PWM alterna rapidamente entre 0 V e $V_{CC}$;
                \item o \textit{duty cycle} controla o valor médio;
                \item um filtro RC suaviza o sinal;
                \item não possui a mesma precisão de um DAC dedicado.
            \end{itemize}

            \[
                V_{\text{médio}} \approx D \times V_{CC}
            \]

            onde $D$ é o \textit{duty cycle}.
        \end{column}
