\begin{frame}{Protocolos de comunicação}
    MQTT:
    \begin{itemize}
        \item Precisa de um servidor
        \item Publicação de mensagens e inscrição em tópico
        \item Baixo overhead
        \item Pouco tráfego
        \item Conexão persistente para evitar gasto de energia
        \item Comunicação desacoplada
        \item QoS (Qualidade de Serviço)
        \item Reconexão automática
        \item Segurança baixa
        \item Dependência de broker
    \end{itemize}
\end{frame}
