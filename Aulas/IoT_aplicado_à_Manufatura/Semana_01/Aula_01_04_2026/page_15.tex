\begin{frame}{Protocolos de comunicação}
    HTTP:
    \begin{itemize}
        \item Precisa de servidor porque faz requisição
        \item Universal e amplamente suportado
        \item Simoles
        \item Integração com APIs
        \item Muito seguro
        \item Muitos dados
        \item Infraestutura pronta
        \item Alto overhead
        \item Maior consumo de energia
        \item Maior latência
        \item Sincronismo
    \end{itemize}
\end{frame}
