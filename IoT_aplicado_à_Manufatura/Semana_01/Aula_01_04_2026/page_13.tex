\begin{frame}{Protocolos de comunicação}
    Wi-Fi:
    \begin{itemize}
        \item 2.4 GHz (maior alance), 5 GHz (mais rápido) e mais recentemente 6 GHz (latência zero)
        \item Access Point ou Station
        \item Interceptação de dados (sniffing)
        \item Muitos dispositivos utilizam
        \item Não determinístico
        \item Alto consumo de energia
        \item Depende de obstáculos
        \item Atenuação em ambiente industrial
        \item Escalabilidade limitada
        \item Simples implementação
        \item Depende de Infraestrutura (roteador e configurações)
    \end{itemize}

    Opções: Zigbee, LoRa, BLE mesh
\end{frame}
