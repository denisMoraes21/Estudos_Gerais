\begin{frame}[fragile]{Principais desafios}
    Ambientes hostis:
    \begin{itemize}
        \item Riscos: calor excessivo, umidade, vibração, poeira, EMI/EMC, produtos químicos
        \item Hardware industrial: -40°C a 85°C, blindagem, filtros, aterramento, proteção contra surto
        \item Proteção física: IP65 (poeira + jato d'água), IP67 (submersão)
        \item Conectores industriais: evitar USB/RJ45, usar M12 (selado)
        \item Comunicação robusta: RS485 (melhor contra ruído, tolerante à falhas)
    \end{itemize}
\end{frame}
