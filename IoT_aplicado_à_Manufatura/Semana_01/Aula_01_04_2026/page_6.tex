\begin{frame}{Principais desafios}
    Segurança:
    \begin{itemize}
        \item Riscos: acesso não autorizado, manipulação de dados, ataques DoS
        \item Confidencialiades: criptografia, VPN
        \item Integridade: Hash, CRC, assinatura digital, Acknowledge
        \item Alta disponibilidade: proteção contra ataques
        \item Autenticação forte: certificado digital, tokens, controle de acesso
        \item Segmentação de rede: industrial e corporativa.
        \item Monitoramento: logs, alertas de tempo real, detecção de intrusão
        \item Atualizações: evitar novo flash por hardware e somnete por OTA.
    \end{itemize}
\end{frame}
