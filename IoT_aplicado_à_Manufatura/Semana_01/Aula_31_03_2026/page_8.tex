\begin{frame}[fragile]{Revisão: Linguagem C}
    Tipos de funções:
    \begin{itemize}
        \item Funções que recebe mais de um parâmetro
    \end{itemize}

\begin{lstlisting}[language=C,basicstyle=\ttfamily\scriptsize]
int sum(int a, int b);
\end{lstlisting}

    \begin{itemize}
        \item Funções que não recebe nenhum parâmetro e tem retorno
    \end{itemize}

\begin{lstlisting}[language=C,basicstyle=\ttfamily\scriptsize]
double get_value(void);
\end{lstlisting}

    \begin{itemize}
        \item Função que recebe parâmetro e não tem retorno
    \end{itemize}

\begin{lstlisting}[language=C,basicstyle=\ttfamily\scriptsize]
void print_number(double number);
\end{lstlisting}

    \begin{itemize}
        \item Função que recebe parâmetro e tem retorno
    \end{itemize}

\begin{lstlisting}[language=C,basicstyle=\ttfamily\scriptsize]
unsigned short process_name(char name[]);
\end{lstlisting}

    \begin{itemize}
        \item Função que não recebe parâmetro e não tem retorno
    \end{itemize}

\begin{lstlisting}[language=C,basicstyle=\ttfamily\scriptsize]
void generic_function(void);
\end{lstlisting}
    
        
\end{frame}
