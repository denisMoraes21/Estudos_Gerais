\begin{frame}[fragile]{Comunicação peer to peer com Esp-Now}
    \begin{lstlisting}[language=C, basicstyle=\tiny\ttfamily]    
    void loop() {
        mensagem.id = 1;
        mensagem.valor = random(0, 100);
    
        esp_now_send(peerMAC, (uint8_t *) &mensagem, sizeof(mensagem));
    
        Serial.print("Enviado: ");
        Serial.println(mensagem.valor);
    
        delay(2000);
    }
    \end{lstlisting}
