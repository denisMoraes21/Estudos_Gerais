\begin{frame}{Testes exploratórios}
    Técnicas:
    \begin{itemize}
        \item Test Tours: tour do usuário iniciante, tour de erros, tour de segurança; tour de atalhos.
        \item Particionamento mental: entradas válidas, inválidas, limites, fluxos críticos.
        \item Heurísticas: sua experiência - campos vazios, caracteres especiais, concorrência, timeout, repetição rápida.
        \item Session-Based Testing: sessões curtas com objetivo, tempo, anotações, defeitos.
    \end{itemize}
\end{frame}
