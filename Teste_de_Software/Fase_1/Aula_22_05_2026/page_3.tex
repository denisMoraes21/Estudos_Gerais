\begin{frame}{Tipos de teste}
    Quanto ao objetivo:
    \begin{itemize}
        \item Funcional
        \item Não funcional
    \end{itemize}
    \vspace{1.0cm}
    Quanto à quantidade de informação disponível:
    \begin{itemize}
        \item Teste de caixa preta (black-box)
        \item Teste de caixa branca (white-box)
        \item Teste de caixa cinza (gray-box)
    \end{itemize}

\end{frame}
